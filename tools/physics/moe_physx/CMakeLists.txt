cmake_minimum_required(VERSION 3.21)
project(moe_physx_bake CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT PHYSX_ROOT_DIR)
  message(FATAL_ERROR "Use build.py to obtain the pinned SDK first")
endif()
set(TARGET_BUILD_PLATFORM windows CACHE STRING "" FORCE)
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(PX_GENERATE_GPU_PROJECTS OFF CACHE BOOL "" FORCE)
set(PX_BUILDSNIPPETS OFF CACHE BOOL "" FORCE)
# Official GPU binaries include OmniPVD support. This changes internal virtual
# tables, so the host libraries MUST use the same setting even without recording.
set(PX_BUILDPVDRUNTIME ON CACHE BOOL "" FORCE)
set(PX_FLOAT_POINT_PRECISE_MATH ON CACHE BOOL "" FORCE)
set(NV_USE_STATIC_WINCRT OFF CACHE BOOL "" FORCE)
set(PX_OUTPUT_LIB_DIR "${CMAKE_BINARY_DIR}/sdk_lib" CACHE PATH "" FORCE)
set(PX_OUTPUT_BIN_DIR "${CMAKE_BINARY_DIR}/sdk_bin" CACHE PATH "" FORCE)
add_subdirectory("${PHYSX_ROOT_DIR}/compiler/public" sdk_host)
# The SDK couples host GPU support to compiling CUDA kernels. We use the matching
# official GPU DLL, so retain host GPU code without requiring a CUDA toolchain.
# Do this on CMake targets, without patching the downloaded SDK source.
get_property(sdk_targets DIRECTORY "${PHYSX_ROOT_DIR}/source/compiler/cmake" PROPERTY BUILDSYSTEM_TARGETS)
foreach(t IN LISTS sdk_targets)
  get_target_property(definitions ${t} COMPILE_DEFINITIONS)
  if(definitions)
    list(REMOVE_ITEM definitions DISABLE_CUDA_PHYSX)
    string(REPLACE "PX_SUPPORT_OMNI_PVD=0" "PX_SUPPORT_OMNI_PVD=1" definitions "${definitions}")
    set_property(TARGET ${t} PROPERTY COMPILE_DEFINITIONS "${definitions}")
  endif()
endforeach()
add_executable(moe_physx_bake main.cpp)
target_include_directories(moe_physx_bake PRIVATE "${PHYSX_ROOT_DIR}/include")
target_compile_definitions(moe_physx_bake PRIVATE PX_PHYSX_STATIC_LIB NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(moe_physx_bake PRIVATE PhysXExtensions PhysX PhysXPvdSDK PhysXCooking PhysXCommon PhysXFoundation PVDRuntime)
target_compile_options(moe_physx_bake PRIVATE /W4 /utf-8 /fp:precise)
target_link_options(moe_physx_bake PRIVATE /MAP)
set_target_properties(moe_physx_bake PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
